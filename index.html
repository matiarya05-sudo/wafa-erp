create table if not exists app_security (
  role text primary key,
  pass_code text not null
);
insert into app_security(role,pass_code) values ('admin','CHANGE_ME') on conflict (role) do nothing;
insert into app_security(role,pass_code) values ('staff','CHANGE_ME') on conflict (role) do nothing;
-- اگر پسورد فعلی را نمی‌دانی، از این خط استفاده کن:
update app_security set pass_code='NEW-STRONG-PASS' where role='admin';

🔑 تغییر پسوردها
🔐 پسوردها در سورس HTML ذخیره نمی‌شوند؛ فقط در جدول app_security در Supabase نگهداری می‌شوند.
🧩 SQL ساخت جدول app_security (یک‌بار اجرا کن) 
create table if not exists app_security (
  role text primary key,
  pass_code text not null
);
-- مقدار CHANGE_ME را با پسورد قوی عوض کن (یا از همین صفحه تغییر بده)
insert into app_security(role,pass_code) values ('admin','CHANGE_ME') on conflict (role) do nothing;
insert into app_security(role,pass_code) values ('staff','CHANGE_ME') on conflict (role) do nothing;
-- تغییر فوری پسورد از SQL (اگر پسورد فعلی را نمی‌دانی)
update app_security set pass_code='NEW-STRONG-PASS' where role='admin';
-- دسترسی با کلید publishable
alter table app_security enable row level security;
drop policy if exists app_security_rw on app_security;
create policy app_security_rw on app_security for all using (true) with check (true);
